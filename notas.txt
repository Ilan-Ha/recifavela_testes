import pandas as pd
from PIL import Image
from torch.utils.data import Dataset


class PlasticDataset(Dataset):

    def __init__(self, csv_file, split, transform=None):
        self.data = pd.read_csv(csv_file)

        # Keep only the requested split
        self.data = self.data[self.data["split"] == split]

        self.transform = transform

        # Convert class names to numbers
        self.label_map = {
            "non_PET": 0,
            "PET": 1
        }

    def __len__(self):
        return len(self.data)

    def __getitem__(self, idx):

        row = self.data.iloc[idx]

        image = Image.open(row["image_path"]).convert("RGB")

        label = self.label_map[row["label"]]

        if self.transform:
            image = self.transform(image)

        return image, label

import json
import pandas as pd
from pathlib import Path

with open("raw/plastics/annotations.json", "r") as f:
    annotations = json.load(f)

rows = []

for filename, annotation in annotations.items():

    material = annotation["material"]

    if material == "PET":
        label = "PET"
    else:
        label = "non_PET"

    rows.append({
        "image_path": f"raw/plastics/images/{filename}",
        "label": label
    })

df = pd.DataFrame(rows)
----------
from sklearn.model_selection import train_test_split

train, temp = train_test_split(
    df,
    test_size=0.30,
    stratify=df["label"],
    random_state=42
)

validation, test = train_test_split(
    temp,
    test_size=0.50,
    stratify=temp["label"],
    random_state=42
)

train["split"] = "train"
validation["split"] = "validation"
test["split"] = "test"

df = pd.concat([train, validation, test])

df.to_csv("dataset.csv", index=False)
